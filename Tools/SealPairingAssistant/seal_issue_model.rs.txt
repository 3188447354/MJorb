/// 一条需要以弹窗告知用户的报错：问题 + 原因 + 解决办法。
#[derive(Clone, PartialEq, Eq)]
struct SealIssue {
    /// 去重键：同一个问题只弹一次，直到问题被解决后才允许再次弹出。
    key: String,
    /// 弹窗标题：一句话说明发生了什么。
    title: String,
    /// 触发原因：必要时附上底层返回的原始错误。
    cause: String,
    /// 解决办法：按顺序逐条列出。
    fixes: Vec<String>,
    /// 弹窗上可选的“立即重试”动作。
    retry: SealIssueRetry,
}

/// 弹窗上的重试动作。
#[derive(Clone, Copy, PartialEq, Eq)]
enum SealIssueRetry {
    /// 只能关闭，需要用户手动处理后再点主按钮。
    None,
    /// 重新检测当前设备（环境类问题）。
    Refresh,
    /// 重新生成配对文件并交接给 Seal（配对类问题）。
    Reprime,
}

impl SealIssue {
    fn new(key: &str, title: &str, cause: String, fixes: &[&str], retry: SealIssueRetry) -> Self {
        Self {
            key: key.to_string(),
            title: title.to_string(),
            cause,
            fixes: fixes.iter().map(|step| (*step).to_string()).collect(),
            retry,
        }
    }
}

/// 压缩底层错误文本，避免弹窗被超长调试输出撑爆。
fn seal_error_text(error: impl std::fmt::Display) -> String {
    let text = error.to_string();
    let text = text.trim();
    if text.is_empty() {
        return "设备未返回具体错误信息".to_string();
    }
    if text.chars().count() > 300 {
        return format!("{}…", text.chars().take(300).collect::<String>());
    }
    text.to_string()
}

/// 连不上 usbmuxd：通常是 iTunes / Apple 移动设备服务缺失。
fn seal_issue_usbmuxd(error: impl std::fmt::Display, install_hint: &str) -> SealIssue {
    SealIssue::new(
        "usbmuxd",
        "无法连接 iPhone 服务",
        format!("连接 usbmuxd 失败：{}", seal_error_text(error)),
        &[
            install_hint,
            "安装完成后重启电脑，确保 iTunes 或「Apple 移动设备服务」正在运行。",
            "解锁 iPhone 并重新插拔数据线，然后点按「重新检测」。",
        ],
        SealIssueRetry::Refresh,
    )
}

/// 配对过程失败：按底层错误归类，给出对症的解决办法。
fn seal_issue_pairing_failed(raw: &str) -> SealIssue {
    let lowered = raw.to_ascii_lowercase();
    if lowered.contains("rejected")
        || lowered.contains("declined")
        || lowered.contains("authenticate the user")
    {
        return seal_issue_pairing_rejected();
    }
    if lowered.contains("tunnelservice") || lowered.contains("untrusted tunnel") {
        return SealIssue::new(
            "pairing-tunnel",
            "设备隧道服务未就绪",
            format!("未能连接设备的非信任隧道服务：{}", seal_error_text(raw)),
            &[
                "解锁 iPhone 并保持数据线连接。",
                "确认 iPhone 系统为 iOS 17.4 及以上（低于该版本请使用本机配对）。",
                "重新插拔数据线，然后点按「重新生成配对文件」。",
            ],
            SealIssueRetry::Reprime,
        );
    }
    if lowered.contains("coredeviceproxy") || lowered.contains("core device") {
        return SealIssue::new(
            "pairing-coredevice",
            "无法连接设备核心代理",
            format!("连接 CoreDeviceProxy 失败：{}", seal_error_text(raw)),
            &[
                "解锁 iPhone 并保持数据线连接，不要中途拔线。",
                "在 iPhone 弹窗中选择「信任」，必要时输入锁屏密码。",
                "点按「重新生成配对文件」重试。",
            ],
            SealIssueRetry::Reprime,
        );
    }
    if lowered.contains("rsd") {
        return SealIssue::new(
            "pairing-rsd",
            "RSD 握手失败",
            format!("与设备执行 RSD 握手失败：{}", seal_error_text(raw)),
            &[
                "解锁 iPhone 并保持数据线连接。",
                "重新插拔数据线，确保系统识别到设备。",
                "点按「重新生成配对文件」重试。",
            ],
            SealIssueRetry::Reprime,
        );
    }
    SealIssue::new(
        "pairing-failed",
        "配对文件生成失败",
        format!("设备返回错误：{}", seal_error_text(raw)),
        &[
            "解锁 iPhone 并保持数据线连接，让屏幕保持点亮。",
            "在 iPhone 弹窗中选择「信任」或「允许」。",
            "点按「重新生成配对文件」重试；仍失败请换用原装数据线并更换 USB 接口。",
        ],
        SealIssueRetry::Reprime,
    )
}
