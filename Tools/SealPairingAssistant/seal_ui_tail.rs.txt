        self.ensure_seal_textures(ctx);
        let seal_icon_texture = self.seal_icon_texture.clone();

        let mut pending_selection: Option<(String, UsbmuxdDevice)> = None;
        if let Some(devices) = self.devices.as_ref()
            && !devices.is_empty()
            && (self.selected_device.is_empty() || !devices.contains_key(&self.selected_device))
            && let Some((name, dev)) = devices.iter().next()
        {
            pending_selection = Some((name.clone(), dev.clone()));
        }

        let preview_device_name = pending_selection
            .as_ref()
            .map(|(name, _)| name.clone())
            .unwrap_or_else(|| self.selected_device.clone());
        let selected_device = self
            .devices
            .as_ref()
            .and_then(|devices| devices.get(&preview_device_name))
            .cloned();

        let value_for = |key: &str| -> Option<String> {
            self.device_info.as_ref().and_then(|info| {
                info.iter()
                    .find(|(item_key, _)| item_key == key)
                    .map(|(_, value)| value.clone())
            })
        };

        let device_name = value_for("device_name")
            .filter(|value| !value.is_empty())
            .or_else(|| {
                if preview_device_name.is_empty() {
                    None
                } else {
                    Some(preview_device_name.clone())
                }
            })
            .unwrap_or_else(|| "未连接 iPhone".to_string());
        let ios_version = value_for("ios_version").unwrap_or_else(|| "—".to_string());
        let seal_lockdown_only = seal_ios_supports_remote_pairing(&ios_version) == Some(false);
        let has_ios_version = ios_version != "—";
        let udid = value_for("udid").unwrap_or_else(|| {
            selected_device
                .as_ref()
                .map(|dev| dev.udid.clone())
                .unwrap_or_else(|| "—".to_string())
        });

        let has_device = selected_device.is_some();
        let environment_checking = self.wireless_enabled.is_none()
            || self.dev_mode_enabled.is_none()
            || self.ddi_mounted.is_none();
        let mut readiness_issues: Vec<&str> = Vec::new();
        if matches!(self.wireless_enabled, Some(Err(_))) {
            readiness_issues.push("无法启用无线调试：保持 iPhone 解锁并点按“信任”后重试");
        }
        match &self.dev_mode_enabled {
            Some(Ok(false)) => readiness_issues.push("开发者模式未开启：在 iPhone 设置中开启后重启设备"),
            Some(Err(_)) => readiness_issues.push("无法读取开发者模式：重新连接数据线并确认信任"),
            _ => {}
        }
        if matches!(self.ddi_mounted, Some(Err(_))) {
            readiness_issues.push("开发者支持文件未就绪：保持 iPhone 解锁后重新检测");
        }
        let environment_ready = has_device && has_ios_version && !environment_checking
            && readiness_issues.is_empty();

        let seal_install_result = self.install_res.get("Seal");
        let seal_installing = matches!(seal_install_result, Some(None));
        let seal_installed = matches!(seal_install_result, Some(Some(Ok(_))));
        let seal_failed = matches!(seal_install_result, Some(Some(Err(_))));
        let is_processing = self.pending_seal_install || seal_installing;
        // 17.0–17.3.1 必须生成 Lockdown 文件；版本未知时若沿用上游默认的
        // RemotePairing，会制造一个该系统根本无法使用的文件。因此先读到系统版本再允许生成。
        let can_primary_action = has_device && has_ios_version && !environment_checking && !is_processing;

        let text_primary = Color32::from_rgb(18, 27, 44);
        let text_secondary = Color32::from_rgb(112, 123, 140);
        let divider = Color32::from_rgba_unmultiplied(205, 216, 229, 150);
        let seal_blue = Color32::from_rgb(37, 124, 172);
        let seal_green = Color32::from_rgb(37, 143, 109);
        let seal_red = Color32::from_rgb(182, 78, 73);

        let status_text = if !has_device {
            "等待连接".to_string()
        } else if self.pending_seal_install {
            self.pairing_file_message
                .clone()
                .unwrap_or_else(|| "正在处理…".to_string())
        } else if seal_installing {
            "正在写入 Seal…".to_string()
        } else if seal_installed {
            "完成".to_string()
        } else if seal_failed {
            "写入失败".to_string()
        } else if !readiness_issues.is_empty() {
            format!("需要处理 {} 项", readiness_issues.len())
        } else if environment_checking {
            if ios_version == "—" {
                "正在检测…".to_string()
            } else {
                format!("iOS {ios_version} · 正在检测")
            }
        } else if ios_version == "—" {
            "已就绪".to_string()
        } else {
            format!("iOS {ios_version} · 已就绪")
        };

        let status_color = if seal_installed {
            seal_green
        } else if seal_failed || !readiness_issues.is_empty() || !has_device {
            seal_red
        } else if is_processing {
            seal_blue
        } else {
            text_secondary
        };

        egui::CentralPanel::default()
            .frame(egui::Frame::new().fill(Color32::TRANSPARENT))
            .show(ctx, |ui| {
                let full_rect = ui.max_rect();
                let window_radius = 10.0;
                let title_h = 64.0;

                ui.painter().rect_filled(
                    full_rect,
                    window_radius,
                    Color32::from_rgba_unmultiplied(246, 250, 253, 238),
                );
                ui.painter().rect_stroke(
                    full_rect.shrink(0.5),
                    window_radius,
                    egui::Stroke::new(1.0_f32, Color32::from_rgba_unmultiplied(255, 255, 255, 210)),
                    egui::StrokeKind::Inside,
                );

                let titlebar = egui::Rect::from_min_max(
                    full_rect.min,
                    egui::pos2(full_rect.right(), full_rect.top() + title_h),
                );
                ui.painter().rect_filled(
                    titlebar,
                    0.0,
                    Color32::from_rgba_unmultiplied(255, 255, 255, 75),
                );
                ui.painter().line_segment(
                    [titlebar.left_bottom(), titlebar.right_bottom()],
                    egui::Stroke::new(1.0_f32, Color32::from_rgba_unmultiplied(255, 255, 255, 180)),
                );

                let drag_rect = egui::Rect::from_min_max(
                    full_rect.min,
                    egui::pos2(full_rect.right() - 244.0, full_rect.top() + title_h),
                );
                let drag_response = ui.interact(
                    drag_rect,
                    ui.id().with("seal-titlebar-drag"),
                    egui::Sense::click_and_drag(),
                );
                if drag_response.drag_started() {
                    ctx.send_viewport_cmd(egui::ViewportCommand::StartDrag);
                }

                if let Some(dev) = selected_device.clone() {
                    let refresh_rect = egui::Rect::from_min_size(
                    egui::pos2(full_rect.right() - 208.0, full_rect.top() + 14.0),
                        egui::vec2(36.0, 36.0),
                    );
                    if ui
                        .put(
                            refresh_rect,
                            egui::Button::new(RichText::new("↻").size(24.0).color(text_primary))
                                .frame(false),
                        )
                        .on_hover_text("重新检测当前 iPhone")
                        .clicked()
                    {
                        self.refresh_device_state(dev);
                    }
                }

                let title_icon_rect = egui::Rect::from_min_size(
                    full_rect.left_top() + egui::vec2(24.0, 16.0),
                    egui::vec2(32.0, 32.0),
                );
                if let Some(texture) = &seal_icon_texture {
                    ui.put(title_icon_rect, egui::Image::new((texture.id(), title_icon_rect.size())));
                } else {
                    ui.painter().rect_filled(title_icon_rect, 10.0, Color32::from_rgb(18, 22, 34));
                    ui.painter().text(
                        title_icon_rect.center(),
                        egui::Align2::CENTER_CENTER,
                        "S",
                        egui::FontId::new(18.0, egui::FontFamily::Proportional),
                        seal_blue,
                    );
                }
                ui.painter().text(
                    egui::pos2(title_icon_rect.right() + 12.0, title_icon_rect.center().y + 1.0),
                    egui::Align2::LEFT_CENTER,
                    t!("app_title"),
                    egui::FontId::new(18.0, egui::FontFamily::Proportional),
                    text_primary,
                );

                let control_y = full_rect.top() + 14.0;
                let control_size = egui::vec2(38.0, 36.0);
                let close_rect = egui::Rect::from_min_size(
                    egui::pos2(full_rect.right() - 48.0, control_y),
                    control_size,
                );
                let max_rect = close_rect.translate(egui::vec2(-42.0, 0.0));
                let min_rect = close_rect.translate(egui::vec2(-84.0, 0.0));
                if ui
                    .put(
                        min_rect,
                        egui::Button::new(RichText::new("—").size(22.0).color(text_primary)).frame(false),
                    )
                    .clicked()
                {
                    ctx.send_viewport_cmd(egui::ViewportCommand::Minimized(true));
                }
                if ui
                    .put(
                        max_rect,
                        egui::Button::new(RichText::new("□").size(20.0).color(text_primary)).frame(false),
                    )
                    .clicked()
                {
                    let maximized = ctx.input(|input| input.viewport().maximized.unwrap_or(false));
                    ctx.send_viewport_cmd(egui::ViewportCommand::Maximized(!maximized));
                }
                if ui
                    .put(
                        close_rect,
                        egui::Button::new(RichText::new("×").size(27.0).color(text_primary)).frame(false),
                    )
                    .clicked()
                {
                    ctx.send_viewport_cmd(egui::ViewportCommand::Close);
                }

                let card_width = (full_rect.width() - 120.0).clamp(620.0, 740.0);
                let card_height = (full_rect.height() - title_h - 76.0).clamp(520.0, 560.0);
                let card_rect = egui::Rect::from_center_size(
                    egui::pos2(full_rect.center().x, full_rect.top() + title_h + card_height / 2.0 + 8.0),
                    egui::vec2(card_width, card_height),
                );

                ui.painter().text(
                    card_rect.center_top() + egui::vec2(0.0, 18.0),
                    egui::Align2::CENTER_CENTER,
                    "设备已连接",
                    egui::FontId::new(13.0, egui::FontFamily::Proportional),
                    seal_green,
                );
                ui.painter().text(
                    card_rect.center_top() + egui::vec2(0.0, 58.0),
                    egui::Align2::CENTER_CENTER,
                    "准备配对",
                    egui::FontId::new(28.0, egui::FontFamily::Proportional),
                    text_primary,
                );
                ui.painter().text(
                    card_rect.center_top() + egui::vec2(0.0, 88.0),
                    egui::Align2::CENTER_CENTER,
                    "确认设备状态后，配对凭据将直接交给 Seal。",
                    egui::FontId::new(14.0, egui::FontFamily::Proportional),
                    text_secondary,
                );

                let detail_width = card_width.min(620.0);
                let detail_left = card_rect.center().x - detail_width / 2.0;
                let mode_text = if !has_ios_version {
                    "检测中"
                } else if seal_lockdown_only {
                    "本机配对"
                } else {
                    "远程配对"
                };
                let device_rect = egui::Rect::from_min_size(
                    egui::pos2(detail_left, card_rect.top() + 118.0),
                    egui::vec2(detail_width, 92.0),
                );
                ui.painter().rect_filled(device_rect, 15.0, Color32::from_rgba_unmultiplied(255, 255, 255, 192));
                ui.painter().text(
                    device_rect.left_top() + egui::vec2(28.0, 34.0),
                    egui::Align2::LEFT_CENTER,
                    &device_name,
                    egui::FontId::new(23.0, egui::FontFamily::Proportional),
                    text_primary,
                );
                ui.painter().text(
                    device_rect.left_top() + egui::vec2(28.0, 62.0),
                    egui::Align2::LEFT_CENTER,
                    format!("iOS {ios_version} · {mode_text}"),
                    egui::FontId::new(14.0, egui::FontFamily::Proportional),
                    text_secondary,
                );
                let device_ready = environment_ready && !seal_failed;
                let ready_badge = egui::Rect::from_center_size(
                    device_rect.right_center() + egui::vec2(-48.0, 0.0),
                    egui::vec2(72.0, 34.0),
                );
                ui.painter().rect_filled(
                    ready_badge,
                    17.0,
                    if device_ready { Color32::from_rgb(225, 246, 237) } else { Color32::from_rgb(241, 231, 220) },
                );
                ui.painter().text(
                    ready_badge.center(),
                    egui::Align2::CENTER_CENTER,
                    if device_ready { "已就绪" } else { &status_text },
                    egui::FontId::new(13.0, egui::FontFamily::Proportional),
                    if device_ready { seal_green } else { status_color },
                );
                let detail_top = card_rect.top() + 315.0;
                let detail_rect = egui::Rect::from_min_size(
                    egui::pos2(detail_left, detail_top),
                    egui::vec2(detail_width, 126.0),
                );
                ui.painter().rect_filled(
                    detail_rect,
                    15.0,
                    Color32::from_rgba_unmultiplied(255, 255, 255, 116),
                );
                ui.painter().rect_stroke(
                    detail_rect,
                    15.0,
                    egui::Stroke::new(1.0_f32, Color32::from_rgba_unmultiplied(255, 255, 255, 220)),
                    egui::StrokeKind::Inside,
                );
                let developer_text = match &self.dev_mode_enabled {
                    None if !has_device => "待连接",
                    None => "检测中",
                    Some(Ok(true)) => "已开启",
                    Some(Ok(false)) => "未开启",
                    Some(Err(_)) => "检测失败",
                };
                let wireless_text = match &self.wireless_enabled {
                    None if !has_device => "待连接",
                    None => "检测中",
                    Some(Ok(())) => "已就绪",
                    Some(Err(_)) => "检查失败",
                };
                let support_text = match &self.ddi_mounted {
                    None if !has_device => "待连接",
                    None => "检测中",
                    Some(Ok(())) => "已就绪",
                    Some(Err(_)) => "检查失败",
                };
                let detail_rows = [
                    ("开发者模式", developer_text, matches!(self.dev_mode_enabled, Some(Ok(true)))),
                    ("无线调试", wireless_text, matches!(self.wireless_enabled, Some(Ok(())))),
                    ("开发者支持文件", support_text, matches!(self.ddi_mounted, Some(Ok(())))),
                ];
                for (index, (label, value, ready)) in detail_rows.iter().enumerate() {
                    let y = detail_top + 22.0 + index as f32 * 42.0;
                    ui.painter().circle_filled(
                        egui::pos2(detail_left + 21.0, y),
                        4.0,
                        if *ready { seal_green } else { Color32::from_rgb(212, 153, 81) },
                    );
                    ui.painter().text(
                        egui::pos2(detail_left + 38.0, y),
                        egui::Align2::LEFT_CENTER,
                        *label,
                        egui::FontId::new(15.0, egui::FontFamily::Proportional),
                        text_primary,
                    );
                    ui.painter().text(
                        egui::pos2(detail_rect.right() - 20.0, y),
                        egui::Align2::RIGHT_CENTER,
                        *value,
                        egui::FontId::new(14.0, egui::FontFamily::Proportional),
                        if *ready { seal_green } else { text_secondary },
                    );
                    if index < 2 {
                        ui.painter().line_segment(
                            [egui::pos2(detail_left + 20.0, y + 21.0), egui::pos2(detail_rect.right() - 20.0, y + 21.0)],
                            egui::Stroke::new(1.0_f32, divider),
                        );
                    }
                }

                let progress_top = card_rect.top() + 236.0;
                let progress_gap = 8.0;
                let segment_width = (detail_width - 3.0 * progress_gap) / 4.0;
                let completed_steps = if seal_installed {
                    4
                } else if seal_installing {
                    3
                } else if self.pairing_file.is_some() {
                    3
                } else if environment_ready {
                    2
                } else if has_device && has_ios_version {
                    1
                } else {
                    0
                };
                let current_step = completed_steps.min(3);
                for (index, label) in ["连接", "准备", "配对", "交接"].iter().enumerate() {
                    let x = detail_left + index as f32 * (segment_width + progress_gap);
                    let segment = egui::Rect::from_min_size(
                        egui::pos2(x, progress_top),
                        egui::vec2(segment_width, 7.0),
                    );
                    let color = if index < completed_steps {
                        seal_green
                    } else if index == current_step {
                        seal_blue
                    } else {
                        Color32::from_rgb(212, 223, 230)
                    };
                    ui.painter().rect_filled(segment, 3.5, color);
                    ui.painter().text(
                        egui::pos2(x + segment_width / 2.0, progress_top + 25.0),
                        egui::Align2::CENTER_CENTER,
                        *label,
                        egui::FontId::new(13.0, egui::FontFamily::Proportional),
                        if index <= current_step { text_primary } else { text_secondary },
                    );
                }
                let guidance = if seal_installed {
                    "配对已完成，可以回到 Seal 继续操作。"
                } else if let Some(issue) = readiness_issues.first() {
                    *issue
                } else if !has_device {
                    "用数据线连接并解锁 iPhone，在设备上点按“信任”。"
                } else if !has_ios_version {
                    "正在读取系统版本，随后自动选择配对方式。"
                } else if is_processing {
                    "保持 iPhone 解锁和数据线连接，等待当前步骤完成。"
                } else if seal_failed {
                    "Seal 未收到凭据。保持连接后重试，并在手机上打开 Seal。"
                } else {
                    "按 iOS 版本自动配对，完成后交给 Seal。"
                };
                ui.painter().text(
                    egui::Rect::from_center_size(
                        egui::pos2(card_rect.center().x, detail_rect.bottom() + 18.0),
                        egui::vec2(detail_width, 22.0),
                    ).center(),
                    egui::Align2::CENTER_CENTER,
                    guidance,
                    egui::FontId::new(14.0, egui::FontFamily::Proportional),
                    if seal_failed || !readiness_issues.is_empty() { seal_red } else { text_secondary },
                );

                let button_text = if is_processing {
                    "正在处理…"
                } else if !has_device {
                    "连接 iPhone 后继续"
                } else if !has_ios_version {
                    "正在读取 iOS 版本…"
                } else if environment_checking {
                    "正在检查设备…"
                } else if !readiness_issues.is_empty() {
                    "检查并修复准备项"
                } else if seal_installed || seal_failed {
                    "重新生成配对文件"
                } else {
                    "生成配对并交给 Seal"
                };
                let button_fill = if can_primary_action {
                    seal_blue
                } else {
                    Color32::from_rgb(160, 178, 198)
                };
                let button_rect = egui::Rect::from_center_size(
                    egui::pos2(card_rect.center().x, card_rect.bottom() - 66.0),
                    egui::vec2(420.0, 48.0),
                );
                let primary_response = ui.put(
                    button_rect,
                    egui::Button::new(
                        RichText::new(button_text)
                            .size(17.0)
                            .strong()
                            .color(Color32::WHITE),
                    )
                    .fill(button_fill)
                    .stroke(egui::Stroke::new(0.0_f32, Color32::TRANSPARENT))
                    .corner_radius(8),
                );
                if can_primary_action
                    && primary_response.clicked()
                    && let Some(dev) = selected_device.clone()
                {
                    if !readiness_issues.is_empty() {
                        self.refresh_device_state(dev);
                    } else {
                        self.reset_pairing_state();
                        self.pending_seal_install = true;
                        self.pairing_file_message = Some("正在生成配对文件…".to_string());
                        self.idevice_sender
                            .send(IdeviceCommands::InstalledApps((
                                dev.clone(),
                                self.supported_app_names(),
                            )))
                            .unwrap();
                        self.idevice_sender
                            .send(IdeviceCommands::GeneratePairingFile((
                                dev,
                                seal_mode_for_ios(&ios_version, self.pairing_mode),
                            )))
                            .unwrap();
                    }
                }

                if let Some(message) = &self.pairing_file_message
                    && !is_processing
                    && !seal_installed
                    && !seal_failed
                {
                    ui.painter().text(
                        button_rect.center_bottom() + egui::vec2(0.0, 24.0),
                        egui::Align2::CENTER_CENTER,
                        message,
                        egui::FontId::new(14.0, egui::FontFamily::Proportional),
                        text_secondary,
                    );
                }

                let line_width = card_width.min(610.0);
                let line_y = card_rect.bottom() - 28.0;
                let line_start = card_rect.center().x - line_width / 2.0;
                ui.painter().line_segment(
                    [egui::pos2(line_start, line_y), egui::pos2(line_start + line_width, line_y)],
                    egui::Stroke::new(1.0_f32, divider),
                );

                let udid_y = line_y + 15.0;
                ui.painter().text(
                    egui::pos2(line_start + 8.0, udid_y),
                    egui::Align2::LEFT_CENTER,
                    "UDID",
                    egui::FontId::new(13.0, egui::FontFamily::Proportional),
                    text_secondary,
                );
                let udid_display = if udid.chars().count() > 28 {
                    let prefix: String = udid.chars().take(9).collect();
                    let suffix: String = udid.chars().rev().take(9).collect::<String>().chars().rev().collect();
                    format!("{prefix}…{suffix}")
                } else {
                    udid.clone()
                };
                ui.painter().text(
                    egui::pos2(line_start + 54.0, udid_y),
                    egui::Align2::LEFT_CENTER,
                    &udid_display,
                    egui::FontId::new(14.0, egui::FontFamily::Monospace),
                    text_primary,
                );
                let copy_rect = egui::Rect::from_min_size(
                    egui::pos2(line_start + line_width - 96.0, udid_y - 18.0),
                    egui::vec2(84.0, 36.0),
                );
                if udid != "—"
                    && ui
                        .put(
                            copy_rect,
                            egui::Button::new(
                                RichText::new("复制")
                                    .size(14.0)
                                    .strong()
                                    .color(seal_blue),
                            )
                            .fill(Color32::from_rgba_unmultiplied(255, 255, 255, 156))
                            .corner_radius(8),
                        )
                        .clicked()
                {
                    ui.ctx().copy_text(udid.clone());
                }

                ui.painter().text(
                    egui::pos2(full_rect.center().x, full_rect.bottom() - 20.0),
                    egui::Align2::CENTER_CENTER,
                    "基于 jkcoxson/idevice_pair 0.1.14 二次开发 · Seal 负责界面与安全交接",
                    egui::FontId::new(12.0, egui::FontFamily::Proportional),
                    text_secondary,
                );
            });

        if let Some((dev_name, dev)) = pending_selection.take() {
            self.select_device(dev_name, dev);
        }
    }

    fn clear_color(&self, _visuals: &egui::Visuals) -> [f32; 4] {
        [0.0, 0.0, 0.0, 0.0]
    }
}
